# Deliberately independent of Elegance's CMake targets, vcpkg and source headers.
cmake_minimum_required(VERSION 4.3)
# Use the C driver: a C++ executable would silently supply the missing runtime and
# fail to test what a Rust prebuilt consumer (which never invokes cc.compile()) links.
project(elegance_package_smoke LANGUAGES C)
if (NOT IS_ABSOLUTE "${ELG_PACKAGE_ROOT}")
    message(FATAL_ERROR "ELG_PACKAGE_ROOT must be the relocated installed package's absolute path")
endif ()
add_executable(smoke main.c)
set_property(TARGET smoke PROPERTY C_STANDARD 11)
if (MSVC)
    set_property(TARGET smoke PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
endif ()
file(STRINGS "${ELG_PACKAGE_ROOT}/elegance-link.txt" lines)
set(have_shim OFF)
set(have_elegance OFF)
foreach (line IN LISTS lines)
    if (NOT line MATCHES "^([^ ]+) (.+)$")
        message(FATAL_ERROR "Invalid manifest entry: ${line}")
    endif ()
    set(kind "${CMAKE_MATCH_1}")
    set(value "${CMAKE_MATCH_2}")
    if (kind STREQUAL "static" OR kind STREQUAL "runtime" OR value MATCHES "^lib/")
        if (IS_ABSOLUTE "${value}" OR value MATCHES "(^|/)\\.\\.(/|$)"
                OR NOT EXISTS "${ELG_PACKAGE_ROOT}/${value}")
            message(FATAL_ERROR "Non-relocatable or missing dependency: ${line}")
        endif ()
        set(value "${ELG_PACKAGE_ROOT}/${value}")
    elseif (IS_ABSOLUTE "${value}" OR value MATCHES "::|\\$<")
        message(FATAL_ERROR "Unresolved dependency: ${line}")
    endif ()
    if (kind STREQUAL "static")
        if (value MATCHES "/(lib)?elegance_c\\.(a|lib)$")
            set(have_shim ON)
            target_link_libraries(smoke PRIVATE "$<LINK_LIBRARY:WHOLE_ARCHIVE,${value}>")
        elseif (value MATCHES "/(lib)?elegance\\.(a|lib)$")
            set(have_elegance ON)
            target_link_libraries(smoke PRIVATE "$<LINK_LIBRARY:WHOLE_ARCHIVE,${value}>")
        else ()
            target_link_libraries(smoke PRIVATE "${value}")
        endif ()
    elseif (kind STREQUAL "lib")
        target_link_libraries(smoke PRIVATE "${value}")
    elseif (kind STREQUAL "runtime")
        add_custom_command(TARGET smoke POST_BUILD COMMAND "${CMAKE_COMMAND}" -E copy_if_different
                "${value}" "$<TARGET_FILE_DIR:smoke>")
    elseif (NOT kind STREQUAL "define")
        message(FATAL_ERROR "Unexpected manifest entry: ${line}")
    endif ()
endforeach ()
if (NOT have_shim OR NOT have_elegance)
    message(FATAL_ERROR "Missing static elegance_c or Elegance")
endif ()
if (APPLE)
    set_property(TARGET smoke PROPERTY BUILD_RPATH "@loader_path")
elseif (UNIX)
    set_property(TARGET smoke PROPERTY BUILD_RPATH "$ORIGIN")
endif ()
enable_testing()
add_test(NAME installed-package COMMAND smoke)
