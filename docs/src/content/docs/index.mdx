```cpp
View Counter() {
    const State<int> count(0);
    return VStack(Text([=] { return std::format("Count: {}", count()); }).font(Font::title),
                  Button("Add one", [=] { count += 1; }));
}
```

	<Card title="Declarative" icon="puzzle">
